# 数据库增量迁移方案

## 摘要

从原始 SQL 文件（12.8MB，宝塔可导入）状态，通过执行少量增量 SQL，变为当前本地数据库结构。

增量 SQL 文件极小（< 5KB），完全不受宝塔 50MB 限制。

## 执行步骤

### 第一步：在宝塔面板导入原始 SQL

将 `jeecg-boot.sql`（12.8MB）在宝塔面板中导入 `jeecg-boot` 数据库。此步骤得到 131 张表 + 种子数据。

### 第二步：执行增量 SQL

```sql
-- ============================================
-- 一、student_bill 表结构调整
-- 添加本地开发中新增的 3 个字段
-- ============================================

ALTER TABLE `student_bill` 
  ADD COLUMN `class_name` VARCHAR(50) NULL DEFAULT '' COMMENT '班级名称' AFTER `major_name`,
  ADD COLUMN `college_name` VARCHAR(50) NULL DEFAULT '' COMMENT '学院名称' AFTER `class_name`,
  ADD COLUMN `isbn` VARCHAR(50) NULL DEFAULT NULL COMMENT 'ISBN' AFTER `textbook_name`;

-- ============================================
-- 二、创建 7 个应用视图
-- ============================================

-- 视图1: 班级关联辅导员信息
CREATE OR REPLACE VIEW `v_class_with_counselor` AS 
  SELECT `c`.`id` AS `id`, `c`.`class_name` AS `class_name`, `c`.`class_code` AS `class_code`, 
         `c`.`major_id` AS `major_id`, `c`.`counselor_id` AS `counselor_id`, 
         `co`.`counselor_id` AS `counselor_no`, `c`.`create_time` AS `create_time`, 
         `c`.`update_time` AS `update_time` 
  FROM (`t_class` `c` LEFT JOIN `t_counselor` `co` ON (`c`.`counselor_id` = `co`.`id`));

-- 视图2: 领取详细记录（关联学生、教材、专业、班级）
CREATE OR REPLACE VIEW `v_receive_with_details` AS 
  SELECT `r`.`id` AS `id`, `r`.`receive_operator` AS `receiveOperator`, 
         `st`.`student_id` AS `studentNo`, `st`.`student_name` AS `studentName`, 
         `r`.`subscription_id` AS `subscription_id`, `sub`.`textbook_id` AS `textbook_id`, 
         `tb`.`textbook_name` AS `textbook_name`, `tb`.`isbn` AS `isbn`, 
         `sub`.`major_id` AS `major_id`, `m`.`major_name` AS `majorName`, 
         `m`.`college_id` AS `college_id`, `c`.`college_name` AS `collegeName`, 
         `sub`.`subscription_year` AS `subscriptionYear`, 
         `sub`.`subscription_semester` AS `subscriptionSemester`, 
         `cl`.`class_name` AS `className`, `r`.`receive_status` AS `receiveStatus`, 
         `r`.`receive_time` AS `receiveTime`, `r`.`receive_remark` AS `receiveRemark`, 
         `r`.`create_time` AS `create_time`, `r`.`update_time` AS `update_time` 
  FROM ((((((`t_receive` `r` 
    LEFT JOIN `t_subscription` `sub` ON (`r`.`subscription_id` = `sub`.`id`)) 
    LEFT JOIN `t_major` `m` ON (`sub`.`major_id` = `m`.`id`)) 
    LEFT JOIN `t_college` `c` ON (`m`.`college_id` = `c`.`id`)) 
    LEFT JOIN `t_student` `st` ON (`r`.`receive_operator` = `st`.`id`)) 
    LEFT JOIN `t_textbook` `tb` ON (`sub`.`textbook_id` = `tb`.`id`)) 
    LEFT JOIN `t_class` `cl` ON (`st`.`class_id` = `cl`.`id`));

-- 视图3: 学生账单汇总（按学号分组求和）
CREATE OR REPLACE VIEW `v_student_bill_summary` AS 
  SELECT `b`.`student_id` AS `studentId`, `s`.`student_id` AS `studentNo`, 
         `s`.`student_name` AS `studentName`, `b`.`class_name` AS `className`, 
         `b`.`major_name` AS `majorName`, `b`.`college_name` AS `collegeName`, 
         `b`.`subscription_year` AS `schoolYear`, 
         `b`.`subscription_semester` AS `semester`, 
         SUM(`b`.`discount_price`) AS `totalDiscountPrice` 
  FROM (`student_bill` `b` 
    LEFT JOIN `t_student` `s` ON (`b`.`student_id` = `s`.`student_id`)) 
  GROUP BY `b`.`student_id`, `s`.`student_id`, `s`.`student_name`, 
           `b`.`class_name`, `b`.`major_name`, `b`.`college_name`, 
           `b`.`subscription_year`, `b`.`subscription_semester`;

-- 视图4: 学生账单关联教材ISBN
CREATE OR REPLACE VIEW `v_student_bill_with_isbn` AS 
  SELECT `b`.`id` AS `id`, `b`.`student_id` AS `student_id`, 
         `b`.`major_name` AS `major_name`, 
         `b`.`subscription_year` AS `subscription_year`, 
         `b`.`subscription_semester` AS `subscription_semester`, 
         `b`.`textbook_name` AS `textbook_name`, `b`.`price` AS `price`, 
         `b`.`discount_price` AS `discount_price`, 
         `b`.`subscribe_status` AS `subscribe_status`, 
         `b`.`receive_status` AS `receive_status`, `b`.`remark` AS `remark`, 
         `b`.`create_time` AS `create_time`, `b`.`update_time` AS `update_time`, 
         `t`.`isbn` AS `isbn` 
  FROM (`student_bill` `b` 
    LEFT JOIN `t_textbook` `t` ON (`b`.`textbook_name` = `t`.`textbook_name`));

-- 视图5: 学生关联辅导员信息
CREATE OR REPLACE VIEW `v_student_with_counselor` AS 
  SELECT `s`.`id` AS `student_id`, `s`.`student_id` AS `student_no`, 
         `s`.`student_name` AS `student_name`, `s`.`major_id` AS `major_id`, 
         `s`.`class_id` AS `class_id`, `s`.`status` AS `status`, 
         `s`.`admission_year` AS `admission_year`, 
         `s`.`user_id` AS `student_user_id`, 
         `c`.`id` AS `class_table_id`, `c`.`class_name` AS `class_name`, 
         `c`.`class_code` AS `class_code`, `c`.`major_id` AS `class_major_id`, 
         `c`.`counselor_id` AS `class_counselor_id`, 
         `co`.`id` AS `counselor_table_id`, 
         `co`.`counselor_name` AS `counselor_name`, 
         `co`.`user_id` AS `counselor_user_id` 
  FROM ((`t_student` `s` 
    LEFT JOIN `t_class` `c` ON (`s`.`class_id` = `c`.`id`)) 
    LEFT JOIN `t_counselor` `co` ON (`c`.`counselor_id` = `co`.`id`));

-- 视图6: 征订详细记录
CREATE OR REPLACE VIEW `v_subscription_with_details` AS 
  SELECT `s`.`id` AS `id`, `s`.`student_id` AS `student_id`, 
         `st`.`student_id` AS `studentNo`, `st`.`student_name` AS `studentName`, 
         `s`.`textbook_id` AS `textbook_id`, `tb`.`textbook_name` AS `textbook_name`, 
         `tb`.`isbn` AS `isbn`, `s`.`selection_id` AS `selection_id`, 
         `s`.`major_id` AS `major_id`, `m`.`major_name` AS `majorName`, 
         `m`.`college_id` AS `college_id`, `c`.`college_name` AS `collegeName`, 
         `s`.`subscription_year` AS `subscriptionYear`, 
         `s`.`subscription_semester` AS `subscriptionSemester`, 
         `s`.`subscribe_status` AS `subscribeStatus`, `s`.`remark` AS `remark`, 
         `s`.`subscribe_time` AS `subscribeTime`, 
         `s`.`create_time` AS `createTime`, `s`.`update_time` AS `updateTime` 
  FROM ((((`t_subscription` `s` 
    LEFT JOIN `t_student` `st` ON (`s`.`student_id` = `st`.`id`)) 
    LEFT JOIN `t_textbook` `tb` ON (`s`.`textbook_id` = `tb`.`id`)) 
    LEFT JOIN `t_major` `m` ON (`s`.`major_id` = `m`.`id`)) 
    LEFT JOIN `t_college` `c` ON (`m`.`college_id` = `c`.`id`));

-- 视图7: 教材选用关联ISBN
CREATE OR REPLACE VIEW `v_textbook_selection_with_isbn` AS 
  SELECT `s`.`id` AS `id`, `s`.`major_id` AS `majorId`, 
         `m`.`major_name` AS `majorName`, `s`.`class_id` AS `classId`, 
         `c`.`class_name` AS `className`, `s`.`textbook_id` AS `textbookId`, 
         `t`.`textbook_name` AS `textbookName`, `t`.`isbn` AS `isbn`, 
         `s`.`school_year` AS `schoolYear`, `s`.`semester` AS `semester`, 
         `s`.`selection_status` AS `selectionStatus`, `s`.`remark` AS `remark`, 
         `s`.`create_time` AS `createTime`, `s`.`update_time` AS `updateTime` 
  FROM (((`t_textbook_selection` `s` 
    LEFT JOIN `t_major` `m` ON (`s`.`major_id` = `m`.`id`)) 
    LEFT JOIN `t_class` `c` ON (`s`.`class_id` = `c`.`id`)) 
    LEFT JOIN `t_textbook` `t` ON (`s`.`textbook_id` = `t`.`id`));
```

## 差异清单

| 类别 | 差异项 | 增量 SQL |
|------|--------|----------|
| 表结构 | `student_bill` 缺 3 列 | `ALTER TABLE ... ADD COLUMN` × 3 |
| 视图 | 7 个视图不存在 | `CREATE OR REPLACE VIEW` × 7 |
| 其他 130 张表 | 完全一致 | 无需变更 |
| 数据 | 仅种子数据 | 无需变更（用户只要求结构） |

## 后续扩展

后期如需导入本地业务数据到宝塔，由于数据量大（1.2G），建议：
1. **分表导出**：`mysqldump jeecg-boot t_student t_subscription t_receive --no-create-info > data.sql`
2. **压缩传输**：gzip 压缩后通常能降到原来的 1/10
3. **分批导入**：按表拆分多个 SQL 文件，每个确保 < 50MB
4. **直接连接**：如果宝塔服务器允许远程连接，用 `mysqldump | mysql` 管道直传

## 验证

执行增量 SQL 后，运行以下查询确认：

```sql
-- 验证 student_bill 列数（应为 16）
SELECT COUNT(*) AS column_count FROM information_schema.COLUMNS 
WHERE TABLE_SCHEMA='jeecg-boot' AND TABLE_NAME='student_bill';

-- 验证视图数量（应为 7）
SELECT COUNT(*) AS view_count FROM information_schema.VIEWS 
WHERE TABLE_SCHEMA='jeecg-boot';

-- 验证表数量（应为 131）
SELECT COUNT(*) AS table_count FROM information_schema.TABLES 
WHERE TABLE_SCHEMA='jeecg-boot' AND TABLE_TYPE='BASE TABLE';
```
